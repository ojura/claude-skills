% ---- Heading styles via plain \makeatletter overrides (no titlesec) ----
% Space reserve before headings. Before a heading, \Needspace* (from the
% bundled needspace.sty) breaks the page when less than the heading's reserve
% is left on it: 7, 6 and 5 baselines for levels 1, 2 and 3, the heading with
% its space above and at least two lines of what follows. The space left on
% the page stays empty when the heading moves to the next one, so a larger
% reserve leaves more pages short at the foot. A longtable directly after a
% heading usually needs more room than the reserve before its first row; the
% longtable patch further down moves the heading to the next page with the
% table when the table cannot start on the heading's page.
% The plain \needspace is not used because its stretchable glue and penalty
% -100 make a break before the heading cost less than the breaks in the
% material that follows, so TeX takes it: a heading that directly follows
% another heading then stays alone at the bottom of the page, the space
% before a heading followed by a table or an [H] figure stays empty, and when
% the break is taken after a longtable has started, the table's continuation
% header prints above the heading.
% \if@nobreak is true directly after a heading, so a heading that directly
% follows another one skips its own reserve. Each heading sets its colour
% after \noindent, inside its own paragraph: \color in vertical mode adds a
% colour node before the paragraph, and the \parskip glue after that node is
% a place where the page can break, between the two headings. Without that
% place the two headings stay together with what follows them, and when they
% do not fit, the page breaks before the first one.
\usepackage{needspace}
\makeatletter
\newdimen\report@headpagetotal
\newcommand\headingreserve[1]{\if@nobreak\else\Needspace*{#1}%
  \global\report@headpagetotal\pagetotal\fi}
\renewcommand\section{\headingreserve{7\baselineskip}\@startsection{section}{1}{\z@}%
  {-3.5ex plus -1ex minus -.2ex}%
  {2.0ex plus .2ex}%
  {\normalfont\Large\bfseries\noindent\color{accent}}}
\renewcommand\subsection{\headingreserve{6\baselineskip}\@startsection{subsection}{2}{\z@}%
  {-2.5ex plus -1ex minus -.2ex}%
  {1.2ex plus .2ex}%
  {\normalfont\large\bfseries\noindent\color{accent}}}
\renewcommand\subsubsection{\headingreserve{5\baselineskip}\@startsection{subsubsection}{3}{\z@}%
  {-2.0ex plus -.7ex minus -.2ex}%
  {.8ex plus .2ex}%
  {\normalfont\normalsize\bfseries\noindent\color{accent!75!black}}}
\makeatother

% fvextra @Break (active under breaklines=true) emits an anti-aliasing
% overlap glue (\vspace{-\FV@backgroundcolorboxoverlap}, default 0.25pt)
% AFTER each line's hbox via \FV@bgcoloroverlap (fvextra.sty:3698). That
% glue node sits UPSTREAM of fancyvrb's \penalty\interlinepenalty (which
% goes before the next box). TeX's break-finding sees the glue as a
% feasible candidate at p=0, ignoring the downstream penalty. samepage
% therefore has no teeth on syntax-highlighted blocks.
%
% Why a naive `\nobreak\vspace{-X}` isn't enough:
%   fancyvrb's @i (the first-line dispatcher) wraps line-processing in an
%   extra `\hbox{}`, so the trailing \FV@bgcoloroverlap runs in restricted
%   HMODE. In hmode, `\vspace{-X}` expands to `\vadjust{\vskip-X\vskip 0pt}`:
%   the \vskip nodes get queued for emission into the OUTER vlist AFTER
%   the line's hbox is shipped. A preceding `\nobreak` in hmode emits a
%   penalty INSIDE the hbox, which the page-breaker can't see (penalties
%   inside hboxes don't affect page-breaking). Net result: the \vskip
%   lands in the outer vlist unprotected, and the page-breaker still sees
%   a p=0 candidate immediately after the first verbatim line.
%
% Fix: build the protective penalty + glue as one atomic vlist unit. In
% vmode, emit \penalty\@M then \vskip directly. In hmode (the @i case),
% wrap BOTH inside a single \vadjust so they enter the outer vlist
% together. \vspace's hmode/vmode duality is the trap; bypassing \vspace
% removes it. The overlap remains visually (PDF anti-aliasing trick to
% hide seams between line backgrounds) but the page-breaker never sees
% it as a feasible candidate.
\makeatletter
\def\FV@bgcoloroverlap{%
  \ifx\FV@backgroundcolorboxoverlap\relax
  \else
    \ifvmode
      \penalty\@M
      \vskip-\FV@backgroundcolorboxoverlap\relax
    \else
      \vadjust{\penalty\@M\vskip-\FV@backgroundcolorboxoverlap\relax}%
    \fi
  \fi}
\makeatother

% Long identifiers like applyCorrectionsPolarAndConvertToCartesian are
% atomic tokens that won't break to fit narrow columns. Workaround: the
% pandoc Lua filter (filter.lua) rewrites \texttt{...} content to insert
% break opportunities at separators (::, ., /, () and camelCase / _ / -

% in filter.lua): "/" and spaces outrank a within-identifier "_", so a

% first section that starts on the page, or the section in force when none

% Rule between two body rows of a longtable (filter.lua emits it). longtable's
% \hline, also in colortbl's version, draws its rule twice with a page-break
% opportunity between the two copies, so with \hline a page break between two
% rows prints the rule under the last row of the page and again under the
% header that the next page repeats. \LTrowrule draws one rule in the table's
% rule colour and allows the page break only after it: the page keeps the
% rule under its last row, and the next page starts with the repeated header
% alone. \LT@cols is longtable's column count and \CT@arc@ colortbl's rule
% colour; the rule row is built as longtable builds the rows of \hline.
\makeatletter
\newcommand\LTrowrule{%
  \noalign{\penalty\@M}%
  \multispan\LT@cols{\CT@arc@\leaders\hrule\@height\arrayrulewidth\hfill}\cr
  \noalign{\penalty\z@}}
\makeatother

% ---- longtable on a page that already holds floats ----
% \@colroom is the height left for text on the current page after the floats
% in its top and bottom areas. Each time LaTeX's output routine runs, for a
% float, a marginal note or a page break, it sets \vsize, the height at which
% TeX breaks the page, to \@colroom. longtable's \LT@start sets \@colroom to
% the full column height \@colht when a table starts (longtable 4.19, in TeX
% Live 2023, and 4.27 of 2026-01-28 both do). When a figure already sits at
% the top or bottom of that page and another figure follows the table on the
% same page, the text therefore gets the full page height and the figures
% already placed are added to it: the page overflows by their height
% ("Overfull \vbox ... has occurred while \output is active"), and its last
% lines and footnotes print below the edge of the paper. Only floating
% figures or tables trigger this, for example after
% \floatplacement{figure}{tbp} in the YAML include-before; with the default
% [H] set below, figures are set in the text flow and never enter those areas.
% The patch removes the reset from \LT@start, so the page on which a table
% starts keeps the room its floats take, and sets the full height, less the
% table's foot, only after the page breaks that longtable makes itself,
% because the page that follows such a break starts without floats. A
% \FloatBarrier before the table would not help: it acts on floats that are
% still waiting for a page, and a figure already placed at the top of the
% current page is not one of them. If a later longtable changes this code,
% \patchcmd leaves the macro as it is and the log shows the warning below.
\makeatletter
\def\report@LTfullcolroom{\global\@colroom\@colht
  \ifvoid\LT@foot\else\global\advance\@colroom-\ht\LT@foot\fi}
\def\report@LTnotpatched#1{\PackageWarningNoLine{markdown-latex-report}%
  {longtable not patched (part #1); a page that holds\MessageBreak
   floats can overflow at a longtable}}
\patchcmd{\LT@start}{\global\@colroom\@colht}{}{}{\report@LTnotpatched{1}}
\patchcmd{\LT@output}{\@outputpage\global\vsize\@colroom}%
  {\@outputpage\report@LTfullcolroom\global\vsize\@colroom}{}%
  {\report@LTnotpatched{2}}
\patchcmd{\LT@output}{\@outputpage\global\vsize\@colroom}%
  {\@outputpage\report@LTfullcolroom\global\vsize\@colroom}{}%
  {\report@LTnotpatched{3}}
\makeatother

% ---- a heading stays with a longtable that directly follows it ----
% \LT@start adds \penalty0 before \LTpre, a place where the page can break,
% and when the table's caption, header and first row do not fit on the page,
% it ends the page at that place. Directly after a heading this leaves the
% heading as the last line of its page: the heading reserves above cover the
% heading and two lines, and a table with a caption usually needs more before
% its first row. \if@nobreak is true directly after a heading until \longtable
% clears it, so the first patch records it. For a table that directly follows
% a heading, \LT@start then adds \nobreak instead of \penalty0, and when the
% caption, header and first row do not fit, \report@LTkeepheading adds a kern
% that overfills the page and a \penalty0. TeX then ends the page at its best
% break before the heading, the heading moves to the next page with the
% table, and a second kern there cancels the first. A heading at the top of
% its page has no break before it, so the kern would overfill that page:
% \headingreserve records \pagetotal after its \Needspace*, and when that was
% zero, the heading starts its page and the macro keeps longtable's own break.
\makeatletter
\newif\ifreport@LTafterheading
\newdimen\report@LTkeepdim
\def\report@LTkeepheading{%
  \ifdim\report@headpagetotal>\z@
    \report@LTkeepdim\pagegoal \advance\report@LTkeepdim-\pagetotal
    \advance\report@LTkeepdim\pageshrink \advance\report@LTkeepdim\p@
    \kern\report@LTkeepdim\penalty\z@\kern-\report@LTkeepdim
  \else
    \vfil\break
  \fi}
\def\report@LTkeepnotpatched#1{\PackageWarningNoLine{markdown-latex-report}%
  {longtable not patched (part #1); a heading can end\MessageBreak
   a page above a longtable that directly follows it}}
\pretocmd{\longtable}%
  {\if@nobreak\report@LTafterheadingtrue\else\report@LTafterheadingfalse\fi}{}%
  {\report@LTkeepnotpatched{1}}
\patchcmd{\LT@start}{\endgraf\penalty\z@\vskip\LTpre}%
  {\endgraf\ifreport@LTafterheading\nobreak\else\penalty\z@\fi\vskip\LTpre}{}%
  {\report@LTkeepnotpatched{2}}
\patchcmd{\LT@start}{\vfil\break}%
  {\ifreport@LTafterheading\report@LTkeepheading\else\vfil\break\fi}{}%
  {\report@LTkeepnotpatched{3}}
\makeatother
