\newcommand{\Derivation@layout}[1]{%
  \edef\Derivation@columns{\csname DerivationColumns@#1\endcsname}%
  \edef\Derivation@rows{\csname DerivationRows@#1\endcsname}%
  \ifnum\Derivation@columns<1
    \PackageError{derivations}{Handout columns must be positive}{Check the scene's HandoutLayout.}%
    \def\Derivation@columns{1}%
  \fi
  \ifnum\Derivation@rows<1
    \PackageError{derivations}{Handout rows must be positive}{Check the exported scene metadata.}%
    \def\Derivation@rows{1}%
  \fi
  \setlength{\DerivationPanelWidth}{\linewidth}%
  \setlength{\@tempdima}{\DerivationPanelGap}%
  \@tempcnta=\Derivation@columns\relax
  \advance\@tempcnta by -1
  \multiply\@tempdima by \@tempcnta
  \addtolength{\DerivationPanelWidth}{-\@tempdima}%
  \divide\DerivationPanelWidth by \Derivation@columns\relax
  \setlength{\DerivationPanelHeight}{\csname DerivationHeightFraction@#1\endcsname\textheight}%
  \setlength{\@tempdima}{\csname DerivationRowGap@#1\endcsname mm}%
  \@tempcnta=\Derivation@rows\relax
  \advance\@tempcnta by -1
  \multiply\@tempdima by \@tempcnta
  \addtolength{\DerivationPanelHeight}{-\@tempdima}%
  \divide\DerivationPanelHeight by \Derivation@rows\relax
  \ifdim\DerivationPanelWidth>0pt\else
    \PackageError{derivations}{The panel gaps consume the handout width}{Reduce DerivationPanelGap or the column count.}%
  \fi
  \ifdim\DerivationPanelHeight>0pt\else
    \PackageError{derivations}{The row gaps consume the handout height}{Reduce the row gap or increase max_height_fraction.}%
  \fi
}
