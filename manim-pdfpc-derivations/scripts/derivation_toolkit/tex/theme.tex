\newcommand{\Derivation@requiredcolor}[1]{%
  \@ifundefinedcolor{\Derivation@palette#1}{%
    \PackageError{derivation-theme}{Missing palette colour \Derivation@palette#1}%
      {Define the six palette roles for both Light and Dark in DerivationPaletteFile.}%
  }{%
    \colorlet{derivation#1}{\Derivation@palette#1}%
  }%
}

  }%

\newcommand{\Derivation@legacygraphic}[2]{%
  \ifDerivationDark
    {\setlength{\fboxsep}{0pt}\colorbox{white}{\includegraphics[#1]{#2}}}%
  \else
    \includegraphics[#1]{#2}%
  \fi
}
\@ifclassloaded{beamer}{%
  \newcommand<>{\DerivationLegacyGraphics}[2][]{%
    \only#3{\Derivation@legacygraphic{#1}{#2}}%
  }
  \newcommand<>{\DerivationUncoverGraphics}[2][]{%
    \begin{tikzpicture}[baseline=(figure.south)]
      \node[inner sep=0,outer sep=0] (figure) {\Derivation@legacygraphic{#1}{#2}};
      \pgfmathsetmacro{\Derivation@veil}{1-\DerivationCoveredOpacity/100}%
      \alt#3{}{\fill[derivationBackground,opacity=\Derivation@veil]
        (figure.south west) rectangle (figure.north east);}%
    \end{tikzpicture}%
  }
}{%
  \newcommand{\DerivationLegacyGraphics}[2][]{\Derivation@legacygraphic{#1}{#2}}
  \newcommand{\DerivationUncoverGraphics}[2][]{\Derivation@legacygraphic{#1}{#2}}
}
