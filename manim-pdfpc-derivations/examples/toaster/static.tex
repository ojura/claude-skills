\begin{frame}{Trial ratios}
  Let $b_i>0$ be the baseline pop time and $o_i>0$ the optimised pop time.
  \[r_i=\frac{b_i}{o_i}.\]
  \uncover<2->{A ratio above one means the optimised trial took less time.}
\end{frame}
\begin{frame}{A typical factor}
  The geometric mean preserves the product of the trial ratios:
  \[S^n=\prod_{i=1}^n r_i,\qquad S=\left(\prod_{i=1}^n r_i\right)^{1/n}.\]
\end{frame}
